#========================Epoll实现TCP并发服务器模型========================#

sfd=socket();

bind();

listen();

//创建epoll实例
int epfd=epoll_create();

struct epoll_event ev;
ev.events=EPOLLIN;
ev.data.fd=sfd;

//将sfd加入到epoll树中
epoll_ctl(epfd,EPOLL_CTL_ADD,sfd,&ev);

struct epoll_event evs[1024];
int size=sizeof(evs)/sizeof(evs[0]);

while(1)
{
    //检测文件描述符是否就绪
    int num=epoll_wait(epfd,evs,size,-1);

    //如果检测到sfd文件描述符就绪，即有客户端发来连接请求
    int newfd=accept();

    //将newfd加入到epoll树中
    struct epoll_event ev;
    ev.events=EPOLLIN;
    ev.data.fd = newfd;
    
    epoll_ctl(epfd,EPOLL_CTL_ADD,newfd,&ev);

    //如果检测到newfd文件描述符就绪，即有客户端发来消息
    int res=recv();
    if(res==0)
    {
        //将该文件描述符从数组中删除
        epoll_ctl(epfd,EPOLL_CTL_DEL,newfd,NULL);

        close(newfd);
    }else if(res>0)
    {
        send();
    }
}
